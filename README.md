# Zain Hackathon Entry
## Payment Agentic System
This project is an entry to Zain's hackathon, as of now, this is a prototype to be submitted at 2026-9-28.  

## How To Run It

You need python installed, then:

```

pip install uv # install the package manager
uv sync # Install the dependencies
uv run fastapi dev # Run the app

```

The script that was used to populate the database with mocked data, generated by claude:

```Bash

#!/usr/bin/env bash
#
# seed_fake_data.sh — populate the Payment Agentic System DB with fake data
#
# Usage:
#   BASE_URL=http://localhost:8000 ./seed_fake_data.sh
#
# Requires: curl, jq
set -euo pipefail

BASE_URL="${BASE_URL:-http://localhost:8000}"

echo "== Seeding against $BASE_URL =="

# ---------------------------------------------------------------------------
# helpers
# ---------------------------------------------------------------------------

sign_up() {
  # $1 = username, $2 = password, $3 = fake card token
  curl -sS -X POST "$BASE_URL/api/sign_up" \
    -H "Content-Type: application/json" \
    -d "{\"username\": \"$1\", \"password\": \"$2\", \"card_token\": \"$3\"}"
}

get_user_id() {
  # $1 = access_token, $2 = username to look up
  curl -sS -X GET "$BASE_URL/api/users/username/$2" \
    -H "Authorization: Bearer $1" | jq -r '.id'
}

add_acquaintance() {
  # $1 = access_token, $2 = acquaintance_name, $3 = notes, $4 = username
  curl -sS -X POST "$BASE_URL/api/acquaintances" \
    -H "Authorization: Bearer $1" \
    -H "Content-Type: application/json" \
    -d "{\"acquaintance_name\": \"$2\", \"notes_on_acquaintance\": \"$3\", \"username\": \"$4\"}"
}

make_request() {
  # $1 = access_token, $2 = amount, $3 = receiver_id
  curl -sS -X POST "$BASE_URL/api/transfer/make_request" \
    -H "Authorization: Bearer $1" \
    -H "Content-Type: application/json" \
    -d "{\"amount\": $2, \"receiver_id\": $3}"
}

confirm_request() {
  # $1 = access_token, $2 = request_id
  curl -sS -X POST "$BASE_URL/api/transfer/confirm_request/$2" \
    -H "Authorization: Bearer $1"
}

# ---------------------------------------------------------------------------
# 1. create fake users
# ---------------------------------------------------------------------------

echo "-- creating users --"

declare -A USERNAMES=(
  [ali]="ali_hassan"
  [zainab]="zainab_k"
  [omar]="omar.dev"
  [noor]="noor_87"
  [mustafa]="mustafa_t"
)

declare -A PASSWORDS=(
  [ali]="Passw0rd123"
  [zainab]="Passw0rd123"
  [omar]="Passw0rd123"
  [noor]="Passw0rd123"
  [mustafa]="Passw0rd123"
)

declare -A TOKENS
declare -A USER_IDS

for key in ali zainab omar noor mustafa; do
  uname="${USERNAMES[$key]}"
  pass="${PASSWORDS[$key]}"
  card_token="tok_fake_$(openssl rand -hex 6)"

  resp="$(sign_up "$uname" "$pass" "$card_token")"
  token="$(echo "$resp" | jq -r '.access_token // empty')"

  if [ -z "$token" ]; then
    echo "  ! sign up failed for $uname: $resp"
    continue
  fi

  TOKENS[$key]="$token"
  USER_IDS[$key]="$(get_user_id "$token" "$uname")"
  echo "  + $uname -> id=${USER_IDS[$key]}"
done

# ---------------------------------------------------------------------------
# 2. add acquaintances (Arabic preferred names, as the agent would store them)
# ---------------------------------------------------------------------------

echo "-- adding acquaintances --"

# Ali's contact list
add_acquaintance "${TOKENS[ali]}" "زينب"   "صديقة من الجامعة"        "${USERNAMES[zainab]}"  > /dev/null
add_acquaintance "${TOKENS[ali]}" "عمر"    "زميل شغل، يرجعلي فلوس الأكل"  "${USERNAMES[omar]}"   > /dev/null
echo "  + ali's acquaintances added"

# Zainab's contact list
add_acquaintance "${TOKENS[zainab]}" "علي"   "أخوي"                 "${USERNAMES[ali]}"     > /dev/null
add_acquaintance "${TOKENS[zainab]}" "نور"   "جارتنا"                "${USERNAMES[noor]}"    > /dev/null
echo "  + zainab's acquaintances added"

# Omar's contact list
add_acquaintance "${TOKENS[omar]}" "مصطفى" "صديق الطفولة"           "${USERNAMES[mustafa]}" > /dev/null
echo "  + omar's acquaintances added"

# ---------------------------------------------------------------------------
# 3. create some transfer requests (mix of pending + confirmed)
# ---------------------------------------------------------------------------

echo "-- creating transfer requests --"

# Ali -> Zainab, 5000, confirmed
req="$(make_request "${TOKENS[ali]}" 5000 "${USER_IDS[zainab]}")"
req_id="$(echo "$req" | jq -r '.request.id')"
confirm_request "${TOKENS[ali]}" "$req_id" > /dev/null
echo "  + ali -> zainab (5000) confirmed, id=$req_id"

# Zainab -> Noor, 2000, confirmed
req="$(make_request "${TOKENS[zainab]}" 2000 "${USER_IDS[noor]}")"
req_id="$(echo "$req" | jq -r '.request.id')"
confirm_request "${TOKENS[zainab]}" "$req_id" > /dev/null
echo "  + zainab -> noor (2000) confirmed, id=$req_id"

# Omar -> Mustafa, 10000, left pending (not confirmed)
req="$(make_request "${TOKENS[omar]}" 10000 "${USER_IDS[mustafa]}")"
req_id="$(echo "$req" | jq -r '.request.id')"
echo "  + omar -> mustafa (10000) left PENDING, id=$req_id"

# Ali -> Omar, 999999999, left pending to demo insufficient-balance handling
req="$(make_request "${TOKENS[ali]}" 999999999 "${USER_IDS[omar]}")"
echo "  + ali -> omar (999999999) attempted (expect SUM_IS_BIGGER_THAN_BALANCE on confirm): $req"

echo "== done =="

```
